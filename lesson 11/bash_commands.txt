# ==========================================
# TERMINAL: Git Golden Path Execution
# ==========================================

# 1. Initialize Git in the local project
git init

# 2. Create a safe working branch (Instead of using main/master)
git checkout -b feature/homework12

# 3. Link local project to the GitHub Cloud (Replace URL with student's actual URL)
git remote add origin YOUR URL

# 4. after we finish the code Stage all files (except those in .gitignore!)
git add .

# 5. Save the snapshot with a descriptive message
git commit -m "comment that explains what we did"

# 6. Push the branch to the cloud
git push -u origin feature/homework12
